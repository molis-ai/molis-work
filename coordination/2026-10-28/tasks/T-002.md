// node t002-check.mjs specs/repository-anti-corruption/privileges.md：T-002 的验收，没有输出、退出码 0 就是过
import { readFileSync } from "node:fs";
const ok = [/^待做$/, /^待做（/, /^待复核（#/, /^已做（/, /^不迁/, /^不做：/];
const t0 = { "已做，见 §3": "023 026 027 031 038 043 047 052 081 088 089 090 095", "不迁，见 §3": "092", "部分，见 §3": "045 058 078" };
const want = new Map(Object.entries(t0).flatMap(([v, ids]) => ids.split(" ").map((n) => [`PG-${n}`, v])));
// 由「待做」「部分」改成「已做」的几行，开头要对上
const lead = { "PG-002": "已做（#1295", "PG-004": "已做（T1 #1250", "PG-005": "已做（T4", "PG-008": "已做（#1280", "PG-026": "已做（#1083", "PG-057": "已做（ac-kernel", "PG-095": "已做（#1263", "PG-102": "已做（batch-29）", "PG-064": "待做（部分已做，W5-08 #1303、A3 #1294、BL-635 #1326；" };
const errors = [];
let inT0 = false;
readFileSync(process.argv[2], "utf8").split("\n").forEach((line, i) => {
  if (/^## /.test(line)) inT0 = /^## 8\. /.test(line);
  const m = /^\| (PG-\d+) \|/.exec(line);
  if (!m) return;
  const cells = line.split(/(?<!\\)\|/).slice(1, -1).map((c) => c.trim());
  const at = `第 ${i + 1} 行 ${m[1]}`;
  if (inT0) {
    if (want.has(m[1]) && !cells[1].startsWith(want.get(m[1]))) errors.push(`${at}：§8 结论应以「${want.get(m[1])}」开头`);
    return;
  }
  if (cells.length < 10) return;
  if (cells.length !== 10) errors.push(`${at}：表格断开，${cells.length} 格`);
  const status = cells.at(-1);
  const opening = status.slice(0, status.indexOf("）") + 1);
  if (lead[m[1]] && !status.startsWith(lead[m[1]])) errors.push(`${at}：开头应是「${lead[m[1]]}」`);
  if (!ok.some((r) => r.test(status))) errors.push(`${at}：状态写法不在 §1 的几种里：${status.slice(0, 20)}`);
  if (/C 合入|C 线/.test(opening)) errors.push(`${at}：开头还写着 C，应是 #1327`);
  if (/本候选/.test(opening)) errors.push(`${at}：开头还写着「本候选」`);

console.log(errors.join("\n"));
